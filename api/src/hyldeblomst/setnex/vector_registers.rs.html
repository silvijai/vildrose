1//! Setnex vector register-bank storage.
2
3use crate::registers::RegisterFile;
4use vildrose_core::trit::Trit;
5use vildrose_isa::setnex::vector_registers::{VECTOR_REGISTER_COUNT, VectorRegister};
6
7/// The number of trit lanes in one Setnex vector register.
8pub const VECTOR_LANE_COUNT: usize = 27;
9
10/// The value stored in one Setnex vector register.
11pub type VectorWord = [Trit; VECTOR_LANE_COUNT];
12
13/// The architectural vector register bank for a Setnex virtual CPU.
14#[derive(Debug, Clone)]
15pub struct SetnexVectorRegisters {
16    file: RegisterFile<VectorWord, VECTOR_REGISTER_COUNT>,
17}
18
19impl SetnexVectorRegisters {
20    /// Creates a vector register bank with all registers reset to zero.
21    #[must_use]
22    pub const fn new() -> Self {
23        Self {
24            file: RegisterFile::new([[Trit::Z; VECTOR_LANE_COUNT]; VECTOR_REGISTER_COUNT]),
25        }
26    }
27
28    /// Returns the current value of `register`.
29    ///
30    /// # Panics
31    ///
32    /// Panics only if an internal Setnex register invariant is violated.
33    #[must_use]
34    pub fn read(&self, register: VectorRegister) -> VectorWord {
35        self.file
36            .read(register.number() as usize)
37            .expect("Setnex vector register always selects one of 27 slots")
38    }
39
40    /// Writes `value` to `register`.
41    pub fn write(&mut self, register: VectorRegister, value: VectorWord) {
42        let written = self.file.write(register.number() as usize, value);
43        debug_assert!(written);
44    }
45}
46
47impl Default for SetnexVectorRegisters {
48    fn default() -> Self {
49        Self::new()
50    }
51}
