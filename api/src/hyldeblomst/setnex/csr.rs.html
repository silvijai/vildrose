1//! Setnex CSR-bank storage.
2
3use crate::registers::RegisterFile;
4use vildrose_core::word::Word27;
5use vildrose_isa::setnex::csr::{CSR_COUNT, Csr};
6
7/// The architectural CSR bank for a Setnex virtual CPU.
8#[derive(Debug, Clone)]
9pub struct SetnexCsrs {
10    file: RegisterFile<Word27, CSR_COUNT>,
11}
12
13impl SetnexCsrs {
14    /// Creates a CSR bank with all CSRs reset to zero.
15    #[must_use]
16    pub const fn new() -> Self {
17        Self {
18            file: RegisterFile::new([Word27::zero(); CSR_COUNT]),
19        }
20    }
21
22    /// Returns the current value of `csr`.
23    ///
24    /// # Panics
25    ///
26    /// Panics only if an internal CSR invariant is violated.
27    #[must_use]
28    pub fn read(&self, csr: Csr) -> Word27 {
29        self.file
30            .read(csr.index())
31            .expect("Setnex CSR always selects one of 27 slots")
32    }
33
34    /// Writes `value` to `csr`.
35    pub fn write(&mut self, csr: Csr, value: Word27) {
36        let written = self.file.write(csr.index(), value);
37        debug_assert!(written);
38    }
39}
40
41impl Default for SetnexCsrs {
42    fn default() -> Self {
43        Self::new()
44    }
45}
