1//! Setnex control/status register identifiers.
2//!
3//! CSR addresses are T3 (3 trits), providing 27 addressable slots
4//! with integer values in -13..=13. Each CSR is a full T27 word.
5//! Unused slots are reserved and read as zero.
6//!
7//! See setnex-spec §2.2 for the architectural CSR map.
8
9use vildrose_core::trit::Trit;
10use vildrose_core::word::Tribble;
11
12/// The number of architectural CSRs addressable by one tribble.
13pub const CSR_COUNT: usize = 27;
14
15/// A Setnex CSR address.
16///
17/// CSR addresses are balanced-ternary tribbles in the range -13 through +13.
18#[derive(Debug, Clone, Copy, PartialEq, Eq)]
19pub struct Csr(Tribble);
20
21impl Csr {
22    // Positive addresses refer to user code and data.
23    // Negative addresses refer to stack and kernel space
24    // Using three trits for definition, rather than from_int(), since it's const methods
25    /// Address Z Z P (decimal 1): Program counter (T27, word address).
26    pub const PC: Self = Self(Tribble::new([Trit::P, Trit::Z, Trit::Z]));
27
28    /// Address N P Z (decimal 2): Logic mode (see §6).
29    pub const LMODE: Self = Self(Tribble::new([Trit::N, Trit::P, Trit::Z]));
30
31    /// Address Z P Z (decimal 3): Arithmetic flags (see §5.5).
32    pub const FLAGS: Self = Self(Tribble::new([Trit::Z, Trit::P, Trit::Z]));
33
34    /// Address Z P P (decimal 4): Exception program counter.
35    pub const EPC: Self = Self(Tribble::new([Trit::P, Trit::P, Trit::Z]));
36
37    /// Address P N N (decimal 5): Exception cause (T27).
38    pub const ECAUSE: Self = Self(Tribble::new([Trit::N, Trit::N, Trit::P]));
39
40    /// Address P N Z (decimal 6): Exception vector (handler address).
41    pub const EVEC: Self = Self(Tribble::new([Trit::Z, Trit::N, Trit::P]));
42
43    /// Address P N P (decimal 7): Processor status (see §2.4).
44    pub const STATUS: Self = Self(Tribble::new([Trit::P, Trit::N, Trit::P]));
45
46    /// Address P Z N (decimal 8): Saved STATUS on exception entry.
47    pub const ESAVE: Self = Self(Tribble::new([Trit::N, Trit::Z, Trit::P]));
48
49    /// Address P Z Z (decimal 9): Exception trap value (see §8.5).
50    pub const ETVAL: Self = Self(Tribble::new([Trit::Z, Trit::Z, Trit::P]));
51
52    /// Address P Z P (decimal 10): Frame-2 exception PC (nested).
53    pub const EPC2: Self = Self(Tribble::new([Trit::P, Trit::Z, Trit::P]));
54
55    /// Address P P N (decimal 11): Frame-2 exception cause.
56    pub const ECAUSE2: Self = Self(Tribble::new([Trit::N, Trit::P, Trit::P]));
57
58    /// Address P P Z (decimal 12): Frame-2 saved STATUS.
59    pub const ESAVE2: Self = Self(Tribble::new([Trit::Z, Trit::P, Trit::P]));
60
61    /// Address P P P (decimal 13): Frame-2 exception trap value.
62    pub const ETVAL2: Self = Self(Tribble::new([Trit::P, Trit::P, Trit::P]));
63
64    /// Address Z Z N (decimal -1): MPU region index selector.
65    pub const MPU_SELECT: Self = Self(Tribble::new([Trit::N, Trit::Z, Trit::Z]));
66
67    /// Address Z N P (decimal -2): Base address of the selected MPU region.
68    pub const MPU_BASE: Self = Self(Tribble::new([Trit::P, Trit::N, Trit::Z]));
69
70    /// Address Z N Z (decimal -3): Config (size + permissions + valid) of selected region.
71    pub const MPU_CFG: Self = Self(Tribble::new([Trit::Z, Trit::N, Trit::Z]));
72
73    /// Address Z N N (decimal -4): Pending-IRQ bitvector.
74    pub const IPENDING: Self = Self(Tribble::new([Trit::N, Trit::N, Trit::Z]));
75
76    /// Address N P P (decimal -5): Per-line IRQ enable mask.
77    pub const IENABLE: Self = Self(Tribble::new([Trit::P, Trit::P, Trit::N]));
78
79    /// Address N P Z (decimal -6): Per-line IRQ priority.
80    pub const IPRIORITY: Self = Self(Tribble::new([Trit::Z, Trit::P, Trit::N]));
81
82    /// Creates a CSR from a tribble address.
83    ///
84    /// Return `None` unless `address` is in `0..CSR_COUNT`.
85    pub fn new(address: Tribble) -> Self {
86        if address.to_int() < -13 || address.to_int() > 13 {
87            return Self(Tribble::new([Trit::Z, Trit::Z, Trit::Z]));
88        }
89        Self(address)
90    }
91
92    /// Returns the ternary CSR address.
93    #[must_use]
94    pub const fn address(self) -> Tribble {
95        self.0
96    }
97
98    /// Returns the dense storage index for this CSR, in `0..CSR_COUNT`.
99    ///
100    /// # Panics
101    ///
102    /// Panics only if the internal tribble-to-index invariant is violated.
103    #[must_use]
104    pub fn index(self) -> usize {
105        // map integer address -13..=13 to 0..=26
106        let offset = self.0.to_int() + 13;
107        usize::try_from(offset)
108            .ok()
109            .filter(|&index| index < CSR_COUNT)
110            .expect("Setnex CSR address always maps to one of 27 slots")
111    }
112
113    /// Creates a CSR from an integer address in `-13..=13`.
114    #[must_use]
115    pub fn from_int(value: i8) -> Option<Self> {
116        Tribble::from_int(value).ok().map(Self)
117    }
118}
